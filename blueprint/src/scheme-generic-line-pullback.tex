\subsection{Original generic-fibre sections under the actual residue-field map}

\begin{definition}[The whole actual residue-fibre module pullback]
\label{berkovich:actual-residue-module-pullback-iso}
\lean{EnvelopeProperty.Berkovich.SchemeValuedPoint.residueModulePullbackIso}
\leanok
For every actual scheme morphism $g:X\to Y$, original point $x\in X$, and
actual original module sheaf $M$ on $Y$, the actual pullback of the original
$\kappa(gx)$-fibre along the genuine residue-field map is isomorphic to the
original $\kappa(x)$-fibre of the SAME actual pulled-back module $g^*M$.
No valuation, properness, invertibility, or fibre-comparison premise is required.
\end{definition}
\begin{proof}
The canonical residue-field morphisms satisfy the actual whole native square
$\operatorname{Spec}(\kappa(x)\leftarrow\kappa(gx))$ followed by the original
generic map to $Y$ equals the original generic map to $X$ followed by $g$.
Use the genuine named module factor isomorphism for that square and the inverse
native pullback-composition comparison for the original generic map and $g$.
Their actual composite is the asserted whole module isomorphism.
\end{proof}

\begin{lemma}[The whole residue comparison retains every original fibre frame]
\label{berkovich:actual-residue-module-pullback-frame}
\lean{EnvelopeProperty.Berkovich.SchemeValuedPoint.residueModulePullbackIso_frame}
\leanok
\uses{berkovich:actual-residue-module-pullback-iso,alggeom:actual-original-frame-factor-pairing}
For ANY actual original frame of the target residue-fibre module, the whole
residue comparison followed by its genuine transported frame is the actual
residue-field pullback of that frame, with the true structure-unit comparison.
\end{lemma}
\begin{proof}
Apply the genuine whole factor-comparison/frame-pairing identity to the native
residue-field square and the actual pullback-composition comparison. The named
residue-module comparison is precisely that actual factor comparison composed
with the inverse composition comparison, so the established identity applies
literally without a frame-preservation hypothesis.
\end{proof}

\begin{definition}[The actual pullback of an arbitrary original generic-section morphism]
\label{berkovich:actual-residue-section-hom-pullback}
\lean{EnvelopeProperty.Berkovich.SchemeValuedPoint.pullbackResidueSectionHom}
\leanok
\uses{berkovich:actual-residue-module-pullback-iso}
Every original morphism from the structure line to the target residue-fibre
module has an actual pullback to the original residue-fibre module of $g^*M$.
The original generic section need not extend to a global section of $M$ on $Y$.
\end{definition}
\begin{proof}
Pull back the actual original morphism through the original residue-field map,
using the genuine structure-unit inverse. Compose with the preceding entire
actual residue-module comparison. Both are actual native module morphisms.
\end{proof}

\begin{definition}[The actual pullback of every original generic-fibre section]
\label{berkovich:actual-residue-section-pullback}
\lean{EnvelopeProperty.Berkovich.SchemeValuedPoint.pullbackResidueSection}
\leanok
\uses{berkovich:actual-residue-section-hom-pullback}
Every original section of the entire target residue-fibre module has an actual
section pullback to the SAME original pulled-back model module's residue fibre.
No global extension or rational-section-preservation condition is required.
\end{definition}
\begin{proof}
Use the genuine structure-line/global-section equivalence on the actual target
residue-field spectrum to obtain its original section morphism. Apply the
preceding actual residue-section pullback, then evaluate that genuine morphism
at the original unit section. This defines the actual original section pullback.
\end{proof}
