\section{Exact original ray denominator division and rational constants}

\begin{lemma}[Division with the same original sections]
\label{lem:native-tc-nef-fs-ray-denominator-division}
\leanok
\uses{def:native-line-whole-section-fs}
\source{fhx-envelope-20261028:eq:FS-function}
\lean{EnvelopeProperty.NativeLineFubiniStudy.FSData.divideDenominator,
EnvelopeProperty.NativeLineFubiniStudy.FSData.divideDenominator_eval,
EnvelopeProperty.NativeLineFubiniStudy.IsFinFS.divideDenominator}
Let $D$ be actual original ray FS data of class $a$ and $N>0$ a natural
number.  Replace its positive denominator $m$ by $mN$ and keep its
original section degree, every original whole section and every original
weight.  The resulting datum has exact class $a/N$ and entire function
$D/N$.  Rationality and absence of common zeros are preserved.  Hence
an original finite rational FS function $\varphi$ of class $a$ gives
$\varphi/N$ of class $a/N$ on the same original native point space.
\end{lemma}
\begin{proof}
The degree identity is $\deg=m a=(mN)(a/N)$ because $N>0$.
The weighted finite supremum is literally unchanged.  The reciprocal
denominator is $(mN)^{-1}=N^{-1}m^{-1}$, giving the stated entire
function identity, including infinite values.  The rational weights
and all original section norms are unchanged, so the same nonzero
section at every original point witnesses absence of common zeros.
\end{proof}

\begin{lemma}[Recovery of the same rational constant]
\label{lem:native-tc-nef-fs-ray-rational-weight-shift}
\leanok
\uses{def:native-line-whole-section-fs}
\source{fhx-envelope-20261028:eq:FS-function}
\lean{EnvelopeProperty.NativeLineFubiniStudy.FSData.shiftWeights,
EnvelopeProperty.NativeLineFubiniStudy.FSData.shiftWeights_eval,
EnvelopeProperty.NativeLineFubiniStudy.IsFinFS.addRational}
For an actual original FS datum with denominator $m$, adding $m c$ to
each original weight adds $c$ to the entire original function and
retains its exact original class.  If $c$ and the original weights are
rational, the new original weights are rational; the same original
sections witness absence of common zeros.
\end{lemma}
\begin{proof}
Addition by the finite real constant $mc$ commutes with the finite
supremum, including the empty supremum.  Multiplication by the positive
real $m^{-1}$ distributes over this sum, and $m^{-1}mc=c$.
The actual weights $\lambda_q+mc$ are rational, and the underlying
original whole sections are unchanged.
\end{proof}

\begin{remark}
For the prescribed original relation $r\varphi=\varphi_M+c$ with integer
$c$, divide an actual finite FS realization of
$b\varphi_M+\psi$ by $rb$, then use the same rational constant $c/r$.
The resulting whole function is exactly
$\varphi+\psi/(rb)$.  No original action is altered and no original
constant is discarded.
\end{remark}
