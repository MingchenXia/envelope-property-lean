\section{Native sheaf cohomology with the literal original coefficients}

\begin{lemma}[Actual sheaf cohomology and the original scalar section complex]
\label{lem:kn-derived-original-scalar-sheaf-cohomology}
\uses{lem:kn-derived-original-scalar-coefficients}
\lean{EnvelopeProperty.Vanishing.SpecializationDerivedSheafComparison.subsingleton_H_iff_restrict_ordered_zero}
\leanok
For an actual finite ordered affine cover of a separated original
scheme, positive native sheaf cohomology of a quasicoherent module
vanishes if and only if native homology of its actual ordered
section complex, restricted along the canonical affine-base ring
isomorphism, vanishes.  The action is the literal original $R$
action; no cohomology comparison is a hypothesis.
\end{lemma}
\begin{proof}
Apply the proved actual affine-cover sheaf-cohomology/ordered-homology
comparison, then the proved zero-homology equivalence under the
actual coefficient restriction.
\end{proof}
