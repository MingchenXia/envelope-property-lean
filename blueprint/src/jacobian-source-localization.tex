\section{The genuine Jacobian ideal retains the original source}

\begin{lemma}[Multilinear membership from a spanning family]\label{lem:sc0-multilinear-submodule-from-generators}
\lean{EnvelopeProperty.Char0.multilinearMap_mem_submodule_of_span_eq_top}
\leanok
If a family spans a module and a multilinear map takes every tuple from that
family into a submodule, then it takes every tuple into that submodule.
\end{lemma}
\begin{proof}
Compose the multilinear map with the actual quotient map by that submodule.
It vanishes on every tuple of spanning generators. The checked multilinear
extensionality theorem therefore makes it the zero map, which is exactly the
asserted membership for every tuple. Empty tuples are included.
\end{proof}

\begin{lemma}[Actual source localization preserves the genuine image ideal]\label{lem:sc0-jacobian-source-localization}
\lean{EnvelopeProperty.Char0.affineRelativeJacobianIdeal_source_localization}
\leanok
\uses{lem:sc0-multilinear-submodule-from-generators,def:sc0-affine-relative-jacobian-ideal}
Let $S$ be an actual localization of the original source $R$, and let $A$ be
an actual compatible algebra over both. In any actual target differential
basis, the genuine ideal generated by all original $R$-function Jacobians is
exactly the ideal generated by all $S$-function Jacobians.
\end{lemma}
\begin{proof}
Every original $R$-tuple is an actual $S$-tuple, proving one inclusion. In the
other direction, the actual localization theorem says the differentials of
original $R$-functions span $\Omega_{S/k}$ over $S$. Compose the actual map
from $\Omega_{S/k}$ to the target differentials with the actual target-basis
determinant. This is an actual $S$-multilinear map, and on original-generator
tuples its value lies in the original $R$-function ideal by definition.
The preceding spanning lemma therefore puts the determinant of every
$S$-tuple in that original ideal. This proves equality without assuming
global coordinates on $R$.
\end{proof}
