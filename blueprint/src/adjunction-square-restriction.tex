\begin{lemma}[The restriction equation of a native adjunction-unit square]
\label{lem:adjunction-square-from-unit-equation}
\lean{EnvelopeProperty.AdjunctionSquareRestriction.square_of_unit_square}
\leanok
Let $F\dashv G$ be the original adjunction. An equality
$r\circ\operatorname{adj}(a)=u\circ G(t)$ is equivalent to
$F(r)\circ a=F(u)\circ\epsilon\circ t$, with the same functors and counit.
\end{lemma}
\begin{proof}
Apply the inverse of the original adjunction hom-set equivalence. Its
naturality in the source and target gives the displayed equality directly.
\end{proof}
