\section{Passing the \'etale boundary complement to actual stalks}

\begin{lemma}[An \'etale boundary complement remains formally \'etale in a localized stalk]\label{lem:sc0-localized-stalk-etale}
\lean{EnvelopeProperty.Char0.formallyEtale_localized_stalk}
\leanok
Let $R\to A_0\to A$ be a tower of rings, with $A$ an actual localization of
$A_0$. Let $q_0\in A_0$ and let $q$ be its image in $A$. If $R\to A_0[1/q_0]$
is formally \'etale, then $R\to A[1/q]$ is formally \'etale.
\end{lemma}
\begin{proof}
The image of $q_0$ is invertible in $A[1/q]$, so the localization universal
property constructs the actual map $A_0[1/q_0]\to A[1/q]$ commuting with the
original maps. Both $A_0\to A\to A[1/q]$ are localizations and hence formally
\'etale; their composite is formally \'etale. The map $A_0\to A_0[1/q_0]$ is
also formally \'etale. Cancellation along this formally unramified map gives
formal \'etaleness of $A_0[1/q_0]\to A[1/q]$. Compose with the assumed formally
\'etale map from $R$.
\end{proof}

\begin{corollary}[The actual Jacobian stalk is a boundary monomial]\label{cor:sc0-jacobian-localized-stalk-monomial}
\lean{EnvelopeProperty.Char0.isUnit_localized_stalk_relativeDifferentialDet_of_monomial_transform,EnvelopeProperty.Char0.exists_monomial_stalk_affineRelativeJacobianIdeal_of_monomial_transform}
\leanok
\uses{lem:sc0-monomial-transform-localized-stalk,lem:sc0-relative-jacobian-away-unit,lem:sc0-relative-jacobian-prime-monomial}
In the preceding situation, suppose $A$ is a domain and the images of the boundary
functions $p_i$ are prime in $A$. Given actual source \'etale coordinates and a
genuine target differential basis, the actual relative Jacobian image ideal on
$A$ is generated by a monomial in these image parameters.
\end{corollary}
\begin{proof}
The actual resolution and total-transform identity produce formal \'etaleness on
the boundary-localized stalk, hence invertibility of the actual determinant there.
The image of the full boundary product is the product of the image parameters.
The prime-boundary factorization theorem gives a unit times a monomial for this
actual determinant. Its principal ideal is the actual all-functions Jacobian
image ideal, yielding the claim.
\end{proof}

\begin{lemma}[The actual monomial transform produces the localized stalk \'etaleness]\label{lem:sc0-monomial-transform-localized-stalk}
\lean{EnvelopeProperty.Char0.formallyEtale_localized_stalk_of_monomial_transform}
\leanok
\uses{lem:sc0-localized-stalk-etale,lem:sc0-monomial-transform-boundary-etale}
Let an actual resolution map $\pi:Y\to\operatorname{Spec}R$ be isomorphic off
the support of an actual ideal sheaf $J$. On an actual affine chart
$\operatorname{Spec}A_0\to Y$, suppose the total transform is generated by
$\prod_i p_i^{e_i}$. For any actual localization $A$ of $A_0$, the map from $R$
to $A[1/\prod_i \overline p_i]$ is formally \'etale.
\end{lemma}
\begin{proof}
The actual monomial identity produces \'etaleness of
$R\to A_0[1/\prod_i p_i]$. Apply the preceding lemma to the actual localization
$A_0\to A$. Ring homomorphisms commute with finite products, so the image of the
boundary product is exactly the product of the image parameters.
\end{proof}
