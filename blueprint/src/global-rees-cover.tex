\section{Finite Rees covers for actual affine scheme centres}

\begin{lemma}[Nonzero finite generators from an actual blowup point]\label{lem:sc0-affine-centre-finite-generators}
\lean{EnvelopeProperty.Char0.exists_finite_generators_of_blowUp_point}
\leanok
\uses{lem:sc0-affine-global-blowup-iso}
Let $R$ be Noetherian and let $J$ be an actual ideal sheaf on $\operatorname{Spec}R$.
If its blowup has an actual point, the normalized ideal of $J$ has a finite generating
family containing a nonzero member.
\end{lemma}
\begin{proof}
The normalized ideal is finitely generated by Noetherianity. It is nonzero by the actual
affine/global blowup comparison and emptiness of the blowup of zero. If every chosen generator
were zero, their ideal span would be zero, a contradiction.
\end{proof}

\begin{lemma}[Finite ratio covers of actual affine scheme blowups]\label{lem:sc0-global-affine-rees-cover}
\lean{EnvelopeProperty.Char0.exists_affineBlowUp_center_iso_of_span,EnvelopeProperty.Char0.globalReesBlowUpChart,EnvelopeProperty.Char0.isOpenImmersion_globalReesBlowUpChart,EnvelopeProperty.Char0.globalReesBlowUpChart_π,EnvelopeProperty.Char0.globalReesBlowUpOpenCover,EnvelopeProperty.Char0.finite_globalReesBlowUpOpenCover_index}
\leanok
\uses{lem:sc0-affine-global-blowup-iso,lem:sc0-actual-rees-chart-cover,lem:sc0-affine-centre-finite-generators}
Let $C\subseteq K$ be a subalgebra of a field and let $J$ be an actual ideal sheaf on
$\operatorname{Spec}C$. For any finite family $g$ generating its normalized ideal, the actual
scheme blowup $J$ has a finite affine cover with coordinate rings exactly
$C[g_1/g_i,\ldots,g_n/g_i]$ for the nonzero $g_i$. The chart maps to $\operatorname{Spec}C$
are induced by the original subalgebra inclusions. If $C$ is Noetherian and the blowup has a
point, such a generating family has a nonzero member, as required at a Rees tree node.
\end{lemma}
\begin{proof}
Use the actual affine/global comparison to identify the scheme blowup with the affine
Rees-Proj blowup of the span of $g$. Compose every open immersion of
Lemma~\ref{lem:sc0-actual-rees-chart-cover} with this isomorphism. Their images cover because
the original images cover and an isomorphism is surjective. The comparison commutes with
the base morphisms, so the resulting chart morphisms still have the asserted inclusion maps.
The index type is the finite subtype of the original generator indices with nonzero denominator.
The preceding lemma supplies the family and its nonzero member under the stated final condition.
\end{proof}
