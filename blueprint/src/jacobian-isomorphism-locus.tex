\section{Geometric production of the Jacobian unit locus}

\begin{lemma}[A chart inside an actual isomorphism open]\label{lem:sc0-open-chart-inside-iso}
\lean{EnvelopeProperty.Char0.isOpenImmersion_comp_of_isIso_restrict}
\leanok
Let $\pi:Y\to X$ be isomorphic over an open $U\subseteq X$, and let $j:Z\to Y$
be an actual open immersion whose image lies in $\pi^{-1}U$. Then $\pi\circ j$
is an open immersion.
\end{lemma}
\begin{proof}
Factor $j$ through the open immersion $\pi^{-1}U\to Y$. The resulting map is an
open immersion by the universal property of open immersions. Its composite with
the actual isomorphism $\pi^{-1}U\to U$ and the open immersion $U\to X$ is the
original composite, which is therefore an open immersion.
\end{proof}

\begin{lemma}[A localized affine chart inside the isomorphism locus is \'etale]\label{lem:sc0-affine-iso-localization-etale}
\lean{EnvelopeProperty.Char0.isOpenImmersion_specMap_localized_of_isIso_restrict,EnvelopeProperty.Char0.etale_algebraMap_localized_of_isIso_restrict}
\leanok
\uses{lem:sc0-open-chart-inside-iso}
Let $\pi:Y\to\operatorname{Spec}R$ be isomorphic over $U$ and let
$\alpha:\operatorname{Spec}A\to Y$ be an actual affine open chart, whose composite
with $\pi$ is induced by $R\to A$. If $D(q)\subseteq(\pi\circ\alpha)^{-1}U$, then
$R\to A[1/q]$ is \'etale and hence formally \'etale.
\end{lemma}
\begin{proof}
The spectrum of $A[1/q]$ is the actual basic open $D(q)$ in $\operatorname{Spec}A$.
Apply the preceding lemma to its open immersion. The resulting composite to
$\operatorname{Spec}R$ is an open immersion, hence \'etale. The affine scheme/ring
equivalence for \'etaleness identifies this map with the original scalar-tower
map $R\to A[1/q]$.
\end{proof}

\begin{lemma}[Actual geometry supplies the localized Jacobian unit]\label{lem:sc0-jacobian-unit-from-iso-open}
\lean{EnvelopeProperty.Char0.isUnit_localized_relativeDifferentialDet_of_isIso_restrict,EnvelopeProperty.Char0.map_affineRelativeJacobianIdeal_eq_top_of_isIso_restrict}
\leanok
\uses{lem:sc0-affine-iso-localization-etale,lem:sc0-relative-jacobian-away-unit}
In the preceding situation, given actual differential bases on $R$ and $A$, the
actual relative differential determinant is a unit in $A[1/q]$. In actual source
\'etale coordinates the all-functions Jacobian image ideal localizes to the unit
ideal.
\end{lemma}
\begin{proof}
The actual isomorphism-open argument produces formal \'etaleness of the localized
map. Apply the proved differential base-change and unit determinant theorem to
that map; use the actual principal image ideal identity for the last assertion.
\end{proof}


\begin{corollary}[A prime boundary chart gives an actual monomial Jacobian]\label{cor:sc0-prime-boundary-jacobian-from-iso}
\lean{EnvelopeProperty.Char0.exists_monomial_affineRelativeJacobianIdeal_of_isIso_restrict}
\leanok
\uses{lem:sc0-jacobian-unit-from-iso-open,lem:sc0-relative-jacobian-prime-monomial}
In the preceding situation, assume $A$ is a domain and $q=\prod_i p_i$ for prime
boundary parameters $p_i$. In actual source \'etale coordinates the actual
relative Jacobian image ideal is generated by a monomial in the $p_i$.
\end{corollary}
\begin{proof}
The actual isomorphism open supplies the determinant's unit in the away
localization. Apply the prime-boundary image ideal lemma to that produced unit.
\end{proof}
