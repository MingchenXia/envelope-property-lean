\begin{theorem}[Whole original Frobenius base square]
\label{lem:cartier-original-scheme-frobenius-base-square}
\lean{EnvelopeProperty.Cartier.SchemeRelativeFrobenius.baseMap,EnvelopeProperty.Cartier.SchemeRelativeFrobenius.twist,EnvelopeProperty.Cartier.SchemeRelativeFrobenius.base_square}
\leanok
Let $k$ be the original field of prime characteristic $p$, let
$f:X\to\operatorname{Spec}k$ be the original structure morphism,
and let $F_k$ be the genuine absolute Frobenius of that same actual
base field scheme. The actual original absolute Frobenius satisfies
$F_X\circ f=f\circ F_k$, as an equality of entire original scheme
morphisms. The actual geometric Frobenius twist is the genuine
native pullback $X\times_{\operatorname{Spec}k,F_k}\operatorname{Spec}k$.
No selected affine presentation or coefficient-square certificate
is supplied.
\end{theorem}
\begin{proof}
Apply the preceding actual whole absolute Frobenius naturality to
this same original $f$, giving the structure morphism of the base
scheme by its genuine identity. This identifies both entire
original composites, including their original section-ring maps.
Form their actual native scheme pullback. It retains the same
original $X$, the same base field scheme and the same actual
absolute Frobenius base map, rather than replacing them by a
ring-level model.
\uses{thm:cartier-original-entire-scheme-frobenius-naturality}
\end{proof}

\begin{theorem}[The same native base power map and genuine perfect-base projection]

\uses{lem:cartier-original-scheme-frobenius-base-square}
