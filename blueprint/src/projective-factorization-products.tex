\section{The actual finite projective embedding of a model product}
\label{sec:projective-factorization-products}

\begin{lemma}[Projective factorizations of the original product]
\label{lem:projective-factorization-products}
\lean{EnvelopeProperty.ProjectiveFactorizationProducts.pullback}
\leanok
\source{bj22-trivval:sec:tc}
Let $R$ be any commutative ring. If the original maps
$f:X\to\operatorname{Spec}R$ and
$g:Y\to\operatorname{Spec}R$ have finite polynomial
projective factorizations over the same $R$, then their
actual product $X\times_RY\to\operatorname{Spec}R$ has
such a factorization, with exactly its original structural map.
\end{lemma}

\begin{proof}
Choose the actual closed embeddings
$i:X\hookrightarrow\mathbf P^m_R$ and
$j:Y\hookrightarrow\mathbf P^n_R$ from the two original
factorizations. They give the actual product morphism
\[
 X\times_RY\longrightarrow\mathbf P^m_R\times_R\mathbf P^n_R.
\]
This is a closed immersion: it is the composite of the two
base changes of the original closed immersions $i$ and $j$.
Compose it with the genuine Segre closed immersion into
$\mathbf P^{(m+1)(n+1)-1}_R$. The two product projection
equations, the original factorization equations and the
original-ring structural equation of the Segre map show
that its composite with the projective projection is
precisely $\operatorname{pr}_Xf$. Thus this is a finite
polynomial projective factorization of the same map.
\end{proof}

\begin{corollary}[Projectivity of the same common graph]
\label{cor:common-proper-model-projective}
\lean{EnvelopeProperty.CommonProperModelGraph.model_isProjectiveFactorization}
\leanok
\uses{lem:projective-factorization-products, thm:common-proper-model-graph}
\source{bj22-trivval:sec:tc, fhx-envelope-20261028:prop:ample-envelopes}
If the two original models have finite projective factorizations
over the same affine ring, then the same actual common graph
$W$ has a finite projective factorization over that ring.
\end{corollary}

\begin{proof}
The common graph is the original closed subscheme of
$X\times_RY$ defined by the graph-image kernel. Precompose
the preceding actual projective factorization with this
closed immersion. Its structural map is exactly $pf=qg$
by the original whole structural triangle.
\end{proof}
