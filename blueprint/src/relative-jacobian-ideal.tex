\section{The actual affine Jacobian image ideal}

\begin{definition}[The ideal of all source-function Jacobians]\label{def:sc0-affine-relative-jacobian-ideal}
\lean{EnvelopeProperty.Char0.sourceFunctionDifferentialMatrix,EnvelopeProperty.Char0.affineRelativeJacobianIdeal}
\leanok
Let $R\to A$ be a map of $k$-algebras and let $b_A$ be an actual differential
basis of rank $n$ for $A/k$. Its relative Jacobian ideal is generated by the
determinants of the $b_A$-coordinate matrices of
$d(\varphi(w_1)),\ldots,d(\varphi(w_n))$, for every $n$-tuple $w$ of elements of
$R$.
\end{definition}

\begin{lemma}[Every function Jacobian factors through the relative determinant]\label{lem:sc0-function-jacobian-factorization}
\lean{EnvelopeProperty.Char0.sourceFunctionDifferentialMatrix_factor,EnvelopeProperty.Char0.sourceFunctionDifferentialMatrix_det_factor}
\leanok
\uses{def:sc0-relative-differential-determinant}
Given an actual differential basis $b_R$ on $R$, the Jacobian of any function
family $w$ is the relative differential determinant times the image in $A$ of the
determinant of the $b_R$-coordinates of $dw$.
\end{lemma}
\begin{proof}
Expand every $dw_j$ in $b_R$. Apply the actual differential map and then each
$b_A$-coordinate functional. The resulting matrix is the product of the relative
differential matrix and the image of the $b_R$-coordinate matrix of $dw$. Take
determinants of this matrix equality.
\end{proof}

\begin{lemma}[Unit locus from an actual \'etale localization]\label{lem:sc0-relative-jacobian-away-unit}
\lean{EnvelopeProperty.Char0.isUnit_localized_relativeDifferentialDet,EnvelopeProperty.Char0.map_affineRelativeJacobianIdeal_eq_top}
\leanok
\uses{lem:sc0-relative-differential-det-basechange,lem:sc0-relative-differential-det-unit,lem:sc0-affine-jacobian-principal}
Suppose $R\to A[1/q]$ is formally \'etale and actual differential bases of equal
rank on $R$ and $A$ are given. The image of the actual relative determinant in
$A[1/q]$ is a unit. In actual source \'etale coordinates, the relative Jacobian
ideal is therefore the unit ideal after this localization.
\end{lemma}
\begin{proof}
Localization of $A$ is formally \'etale, so differential base change identifies
the determinant's image with the actual relative determinant for $R\to A[1/q]$.
The latter is a unit by the formal \'etale determinant lemma. The principal-ideal
description then gives the assertion for the actual ideal.
\end{proof}

\begin{lemma}[An actual Jacobian ideal supported on a prime boundary is monomial]\label{lem:sc0-relative-jacobian-prime-monomial}
\lean{EnvelopeProperty.Char0.exists_monomial_affineRelativeJacobianIdeal}
\leanok
\uses{lem:sc0-relative-jacobian-away-unit,lem:sc0-affine-jacobian-principal,lem:sc0-unit-away-monomial}
Let $A$ be a domain and $p_i$ a finite family of prime elements. If
$R\to A[1/\prod_i p_i]$ is formally \'etale and $R$ has actual source \'etale
coordinates, its actual Jacobian ideal on $A$ is generated by a monomial in the
$p_i$.
\end{lemma}
\begin{proof}
The ideal is principal with generator the actual relative determinant. The
preceding lemma makes this generator a unit after inverting the boundary
product. Lemma~\ref{lem:sc0-unit-away-monomial} expresses it as a unit times a
boundary monomial. Multiplying a generator by a unit preserves its principal
ideal. The empty boundary has product one, giving the unit ideal.
\end{proof}

\begin{lemma}[The actual Jacobian ideal is generated by the relative determinant]\label{lem:sc0-affine-jacobian-principal}
\lean{EnvelopeProperty.Char0.affineRelativeJacobianIdeal_le_span,EnvelopeProperty.Char0.affineRelativeJacobianIdeal_eq_span,EnvelopeProperty.Char0.affineRelativeJacobianIdeal_eq_top,EnvelopeProperty.Char0.affineRelativeJacobianIdeal_ne_bot_of_common_fractionField}
\leanok
\uses{def:sc0-affine-relative-jacobian-ideal,lem:sc0-function-jacobian-factorization,lem:sc0-etale-differential-basis}
If $z$ is an actual \'etale coordinate family on $R$, the relative Jacobian ideal
is generated by the actual relative determinant in the bases $dz$ and $b_A$.
It is nonzero when $R$ and $A$ have an actual common fraction field and is the
unit ideal when $R\to A$ is formally \'etale.
\end{lemma}
\begin{proof}
The factorization lemma shows that every generating function Jacobian belongs to
the principal ideal of the relative determinant. Conversely use the function
family $w=z$. Its source differential coordinate matrix is the identity, so its
Jacobian is the relative determinant itself. This proves both inclusions of ideals.
Generic nonvanishing and the formally \'etale unit assertion for the determinant
give the corresponding assertions for its principal ideal.
\end{proof}
