\subsection{The whole original integral diagram factors through its genuine generic schematic image}

\begin{definition}[Actual whole image factor from a schematically dense restriction]
\label{scheme:dominant-image-factorization}
\lean{EnvelopeProperty.SchemeDominantImageFactorization.factorThroughImage}
\leanok
Let $a:W\to T$ be an actual schematically dominant morphism and let
$q:T\to X$ be an actual whole original scheme morphism. The SAME original
$q$ factors canonically through the actual scheme-theoretic image
$(a\circ q).\mathrm{image}$ of its actual dense restriction.
\end{definition}
\begin{proof}
The genuine native scheme kernel satisfies
$\ker(a\circ q)=q_*(\ker a)=q_*(0)=\ker q$.
The canonical map from $T$ to the actual scheme image of $q$, followed by
the genuine closed-subscheme inclusion supplied by this ideal equality,
defines the required actual whole morphism. No pointwise containment or
closed-image factorization is presumed.
\end{proof}

\begin{theorem}[Both original whole diagram triangles are retained]
\label{scheme:dominant-image-factorization-triangles}
\lean{EnvelopeProperty.SchemeDominantImageFactorization.factorThroughImage_inclusion,EnvelopeProperty.SchemeDominantImageFactorization.dominant_factorThroughImage}
\leanok
\uses{scheme:dominant-image-factorization}
The actual whole image factor followed by the actual closed image inclusion
is EXACTLY the SAME original $q$. Its restriction along the SAME original
$a$ is EXACTLY the genuine canonical map of $a\circ q$ to its actual image.
\end{theorem}
\begin{proof}
The first equality is the native closed-subscheme inclusion identity
composed with the canonical image factorization identity. For the second,
compose with the SAME genuine closed image inclusion. Both sides become
the SAME actual composite $a\circ q$. The closed immersion is a monomorphism,
so cancellation proves the entire morphism equality.
\end{proof}

\begin{theorem}[Genuine schematic images of integral sources are integral]
\label{scheme:integral-schematic-image}
\lean{EnvelopeProperty.SchemeDominantImageFactorization.image_isIntegral}
\leanok
The actual scheme-theoretic image of ANY actual quasi-compact map from an
integral scheme is integral. No integrality or reducedness certificate for
the image is supplied.
\end{theorem}
\begin{proof}
The genuine canonical map to the actual image is schematically dominant.
Its original source is reduced, so the actual section injections derive
reducedness of the SAME entire image. Its actual topological image is dense
and the source is irreducible. The image of the irreducible whole source,
followed by actual closure, therefore proves irreducibility of the whole
scheme image. A reduced irreducible scheme is integral.
\end{proof}
