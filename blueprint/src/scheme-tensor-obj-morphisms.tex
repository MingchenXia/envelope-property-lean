\section{Original explicit tensor maps and actual line tensor exactness}

\begin{definition}\label{sc0-tensor-obj-morphisms:map}
\lean{EnvelopeProperty.SchemeTensorObjMorphisms.tensorObjMap}
\leanok
For actual original module sheaf maps $a:M\to M'$ and $b:N\to N'$, take the SAME explicit original sheafified tensor map: sheafify the original presheaf tensor $a\otimes b$.
\end{definition}
\begin{proof}
This is the genuine native sheafification functor applied to the actual original presheaf tensor morphism, with exactly its original source and target tensor sheaves.
\end{proof}

\begin{lemma}\label{sc0-tensor-obj-morphisms:natural}
\lean{EnvelopeProperty.SchemeTensorObjMorphisms.tensorObjMap_monoidal}
\leanok
\uses{sc0-tensor-obj-morphisms:map}
The genuine original comparison between the explicit sheafified tensor and native monoidal tensor is natural for ENTIRE original tensor maps.
\end{lemma}
\begin{proof}
Apply the actual sheafification Hom equivalence to compare the entire maps on the SAME original presheaf tensor. Actual tensor induction reduces every original section to an original pure tensor. Naturality of the SAME original sheafification unit computes the explicit map there. The native pure tensor section is precisely the unit image followed by the inverse original tensor comparison. Its proved naturality under the SAME original maps gives the required whole pure tensor equality. Zero and addition extend it to every original tensor, and actual adjunction injectivity proves the entire-map equality. No comparison certificate is assumed.
\end{proof}

\begin{theorem}\label{sc0-tensor-obj-morphisms:mono}
\lean{EnvelopeProperty.SchemeTensorObjMorphisms.mono_tensorObjMap_left}
\leanok
\uses{sc0-tensor-obj-morphisms:natural}
Tensoring an actual monomorphism by the SAME original invertible module preserves monicity for the ORIGINAL explicit sheafified tensor map, with no flatness or exactness certificate.
\end{theorem}
\begin{proof}
The original invertible module and its true sheaf dual already derive an actual equivalence given by native right tensor. Hence this functor preserves monomorphisms. The genuine native braiding compares left tensor to this SAME right tensor, so actual left tensor also preserves monomorphisms. The preceding entire-map naturality expresses the explicit original tensor morphism as its native monoidal counterpart conjugated by the SAME original tensor comparison isomorphisms. Monicity is therefore preserved for the actual explicit map itself.
\end{proof}
