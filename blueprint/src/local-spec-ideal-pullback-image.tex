\section{Whole original ideal pullback images over local spectra}

\begin{lemma}[An actual local-spectrum morphism factors through an affine neighborhood]
\label{lem:actual-local-spectrum-affine-factor}
\lean{AlgebraicGeometry.Scheme.exists_lift_open_of_closedPoint_mem}
\leanok
Let $A$ be an actual original local ring and
$f:\operatorname{Spec}A\to X$ any actual morphism. For every original
affine open neighborhood $U$ of the image of the actual closed
point, $f$ factors through the genuine inclusion $U\hookrightarrow X$.
\end{lemma}
\begin{proof}
The inverse image of $U$ contains the actual closed point. Every
prime of $A$ specializes to that point, so the inverse image is the
whole spectrum. The entire image of $f$ is therefore contained in
$U$. The genuine universal property of the original open immersion
provides an actual factor $g:\operatorname{Spec}A\to U$ with
$g\iota_U=f$.
\end{proof}

\begin{lemma}[Actual local-spectrum ideal pullback is onto the whole original ideal]
\label{lem:actual-local-spectrum-ideal-pullback-surjective}
\lean{AlgebraicGeometry.Scheme.IdealSheafData.localSpec_toModulesComapPullbackMap_appTop_surjective}
\leanok
\uses{lem:actual-local-spectrum-affine-factor,cor:actual-affine-ideal-pullback-comparison-surjective}
For every original ideal sheaf $I$ on an arbitrary original scheme
and every actual morphism from an original local spectrum, the
canonical ideal-module pullback comparison is surjective on the
entire original global section module.
\end{lemma}
\begin{proof}
Choose an actual affine neighborhood of the image of the original
closed point and the preceding genuine factorization $f=g\iota_U$.
The actual ideal-module comparison for the original open immersion
is an isomorphism, and its actual pullback by $g$ is still an
isomorphism. The actual comparison on $g$ is surjective on global
sections by the genuine affine result. Their composite is therefore
surjective on the entire original global section module. The true
canonical comparison composition equation identifies that composite
with the comparison for $g\iota_U$ precomposed by the native
pullback composition isomorphism and postcomposed by the native
whole geometric ideal equality isomorphism. These actual global
isomorphisms can be cancelled without changing surjectivity.
Substitute the original factorization $g\iota_U=f$.
\end{proof}

\begin{corollary}[Entire local-spectrum coefficient image]
\label{cor:actual-local-spectrum-ideal-inclusion-image}
\lean{AlgebraicGeometry.Scheme.IdealSheafData.localSpecPullbackInclusionRange_eq}

\begin{proof}

\end{proof}

\begin{corollary}[Actual frame coefficient generates the whole pullback ideal]
\label{cor:actual-local-spectrum-framed-ideal-principal}

\end{corollary}
\begin{proof}

linearity identifies its image with $c$ times the image of
$e^{-1}(1)$. Thus the entire inclusion range lies in the stated

the principal ideal lies in the range. Apply the preceding entire
image equality. No nonvanishing assertion is needed.
\end{proof}
