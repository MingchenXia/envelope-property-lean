\section{The actual valuative multiplier ideal}

\begin{definition}[The intrinsic source Jacobian ideal in an actual divisorial DVR]\label{def:sc0-intrinsic-source-dvr-jacobian}
\lean{EnvelopeProperty.Char0.intrinsicDVRSourceJacobianIdeal}
\leanok
\uses{lem:sc0-actual-finite-dvr-coordinates,thm:sc0-dvr-jacobian-presentation-independent}
For an actual essentially finite-type divisorial DVR $C\subseteq V\subseteq K$,
choose an actual uniformizer and finite lifted residue-transcendence basis,
and form the ideal generated by the Jacobians of all original source-function
tuples. Denote it by $\operatorname{Jac}_V(C)$. The result is independent of
all choices, including the finite number and enumeration of coordinates.
\end{definition}

\begin{definition}[The valuative multiplier ideal]\label{def:sc0-valuative-multiplier-ideal}
\lean{EnvelopeProperty.Char0.valuativeMultiplierIdeal}
\leanok
\uses{def:sc0-intrinsic-source-dvr-jacobian,def:sc0-dvr-multiplier-contribution}
Let $C$ be a domain of characteristic zero with actual function field $K$,
and let $\mathfrak a\subset C$ be nonzero. For $q\in\mathbf Q$, set
\[
 \mathcal J_C(\mathfrak a^q)
 =\bigcap_V\bigl(C\cap\mathfrak m_V^{\max\{0,\lfloor q\operatorname{ord}_V(\mathfrak aV)\rfloor-
       \operatorname{ord}_V\operatorname{Jac}_V(C)\}}\bigr),
\]
where $V$ ranges over every actual essentially finite-type DVR between $C$
and $K$ whose fraction field is $K$. This intersection is an actual ideal
of the original ring $C$. On smooth sources it is the valuative definition
of the multiplier ideal; no finite determining set of valuations is asserted.
\end{definition}

\begin{theorem}[Exact divisorial membership]\label{thm:sc0-valuative-multiplier-membership}
\lean{EnvelopeProperty.Char0.mem_valuativeMultiplierIdeal_iff}
\leanok
\uses{def:sc0-valuative-multiplier-ideal,lem:sc0-dvr-multiplier-contribution-membership}
An original source element $f$ belongs to $\mathcal J_C(\mathfrak a^q)$ if
and only if, for every actual DVR in the defining intersection, either $f=0$
or
\[
 q\operatorname{ord}_V(\mathfrak aV)
 <\operatorname{ord}_V(f)+1+\operatorname{ord}_V\operatorname{Jac}_V(C).
\]
\end{theorem}
\begin{proof}
Membership in an intersection is membership in every constituent ideal.
Apply the exact strict-discrepancy membership theorem to each actual DVR
contribution. Its coordinate-independence identifies the source Jacobian
order with the intrinsic order used above, in every finite presentation.
\end{proof}

\begin{theorem}[The original ideal lies in its multiplier ideal at exponent one]\label{thm:sc0-original-ideal-multiplier-ideal}
\lean{EnvelopeProperty.Char0.le_valuativeMultiplierIdeal_one}
\leanok
\uses{def:sc0-valuative-multiplier-ideal,lem:sc0-original-ideal-multiplier-contribution}
The original ideal satisfies $\mathfrak a\subseteq\mathcal J_C(\mathfrak a)$.
\end{theorem}
\begin{proof}
Every original element of $\mathfrak a$ belongs to its exponent-one
contribution for each actual DVR, by the contribution inclusion theorem.
It therefore belongs to their intersection.
\end{proof}
