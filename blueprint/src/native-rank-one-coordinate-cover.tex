\section{The original coordinates cover the whole native rank-one model}

\begin{lemma}[Original integrality forces the coordinate radical containment]
\label{lem:fc-native-rank-one-coordinate-radical}
\leanok
\lean{EnvelopeProperty.MultiModel.nativeRankOne_coordAlg_eq, EnvelopeProperty.MultiModel.nativeRankOneCoordinateIdeal, EnvelopeProperty.MultiModel.nativeRankOne_positive_mem_coordinateRadical, EnvelopeProperty.MultiModel.nativeRankOne_irrelevant_le_coordinateRadical}
\uses{lem:fc-native-rank-one-homogeneous-polynomial, def:gl-model}
Every actual positive-degree homogeneous element of the original
rank-one algebra belongs to the radical of the ideal generated by its
SAME original degree-one coordinates.
\end{lemma}
\begin{proof}
Write the element as its original coefficient times the monomial of its
positive degree $d$. Apply the original model's integrality axiom to get
a monic equation over the original coordinate algebra, of degree $N>0$.
Extract its degree-$Nd$ coefficient. The coefficient multiplying the
$i$th power has original positive degree $(N-i)d$. Every such positive
coefficient in the coordinate algebra is a sum of original coordinates
times coefficients of degree one less, by the actual homogeneous
coordinate-algebra expansion. Lift that identity back to the original
algebra. It puts the $N$th power of the original element in the actual
coordinate ideal, proving the radical containment.
\end{proof}

\begin{theorem}[The actual original coordinate opens cover the whole native scheme]
\label{thm:fc-native-rank-one-coordinate-cover}
\leanok
\lean{EnvelopeProperty.MultiModel.nativeRankOne_iSup_coordinateOpen}
\uses{lem:fc-native-rank-one-coordinate-radical, lem:fc-native-rank-one-model-charts}
The finitely many basic opens of the SAME original coordinates cover the
whole actual native rank-one scheme.
\end{theorem}
\begin{proof}
The irrelevant ideal is spanned by the original positive homogeneous
pieces. The preceding containment puts it in the radical of the original
coordinate ideal. A homogeneous prime containing every original
coordinate would therefore contain the irrelevant ideal, contrary to
its being a point of the actual Proj. Thus every actual native point is
in one of the stated original coordinate opens.
\end{proof}
