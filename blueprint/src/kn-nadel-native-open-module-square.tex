\begin{lemma}[The original module square under native open restriction]
\label{lem:kn-nadel-native-open-module-square}
\lean{EnvelopeProperty.Vanishing.NadelGlobalCanonicalTrace.restrict_originalModuleSquareMap}
\leanok
The original module-square map for an open cartesian square has the exact native
restriction--pushforward comparison as its restriction adjoint, for every module.
\end{lemma}
\begin{proof}
Use the original restriction--pullback unit identity and naturality on every open.
The remaining restriction morphisms coincide in the opens poset.
\end{proof}
