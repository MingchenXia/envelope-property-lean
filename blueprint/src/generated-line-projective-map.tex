\begin{theorem}[Actual affine original restriction compatibility]
\label{thm:fc-generated-line-projective-map-local-compatibility}
\leanok
\lean{EnvelopeProperty.GeneratedLineProjectiveMap.localMap, EnvelopeProperty.GeneratedLineProjectiveMap.localMap_restrict, EnvelopeProperty.GeneratedLineProjectiveMap.chartMap, EnvelopeProperty.GeneratedLineProjectiveMap.chartMap_naturality}
\uses{thm:fc-generated-line-projective-affine-naturality, thm:fc-original-generator-local-frame-restriction}
On any actual original framed affine open use the original finite
family's actual affine projective tuple map. Under a literal original
affine inclusion, restrict the original frame. The resulting entire
map is the original map composed with the same inclusion. Replacing
the restricted frame by any actual original frame leaves the map
unchanged.
\end{theorem}
\begin{proof}
The same original generator coefficient on the smaller open is the
actual top-section restriction of its original coefficient. Actual
finite relabeling and zero padding preserve this entire equality.
Native affine-spectrum and unit-ideal Rees tuple naturality therefore
give the whole scheme equality. Genuine whole frame independence
handles any other actual original frame on the smaller open.
\end{proof}

\begin{definition}[The whole original finite-generator projective map]
\label{def:fc-generated-line-whole-projective-map}
\leanok
\lean{EnvelopeProperty.GeneratedLineProjectiveMap.projectiveMap, EnvelopeProperty.GeneratedLineProjectiveMap.projectiveMap_restrict}
\uses{thm:fc-generated-line-projective-map-local-compatibility, thm:fc-original-line-framed-affine-cover}

\begin{proof}
The preceding actual restriction equalities are exactly the

colimit. Apply its genuine gluing construction to obtain the whole
original map. Its restriction to each original framed affine open is
the actual original local tuple map, by the native colimit identity.
\end{proof}

\begin{proof}

\end{proof}
