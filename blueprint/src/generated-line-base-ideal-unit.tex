\section{The whole base ideal of an actually generated line is the unit ideal}

\begin{lemma}[Original generating sections retain their actual top sections]
\label{lem:generated-line-top-sections}
\leanok
\lean{EnvelopeProperty.GeneratedLineBaseIdealUnit.topSection,EnvelopeProperty.GeneratedLineBaseIdealUnit.sectionHom_topSection}
For an actual generating family $G$ of a module sheaf $L$,
evaluate each original compatible section at the top open.
The resulting original global section defines precisely the
original basis inclusion followed by the original free evaluation map.
\uses{def:actual-generating-section-hom}
\end{lemma}
\begin{proof}
A compatible section is exactly the family of restrictions
of its value at the top open, by its compatibility for each
inclusion into the top. Apply the actual section-to-morphism
bijection to this equality and to the original free evaluation.
Its defining basis formula gives the asserted original morphism.
\end{proof}

\begin{lemma}[Actual generated-line tensor evaluation is epimorphic]
\label{lem:generated-line-dual-evaluation-epi}
\leanok
\lean{EnvelopeProperty.GeneratedLineBaseIdealUnit.familyEvaluation_generators,EnvelopeProperty.GeneratedLineBaseIdealUnit.familyEvaluation_generators_epi}
If $L$ is invertible and $G$ genuinely generates $L$, the
actual dual evaluation of these same original global sections
is the original free evaluation tensored by $L^\vee$, followed
by the original pairing $L\otimes L^\vee\to\mathcal O_X$.
It is an epimorphism, with no finiteness or nonemptiness condition.
\uses{lem:generated-line-top-sections,prop:actual-invertible-section-family-image}
\end{lemma}
\begin{proof}
Tensoring by the genuine inverse line preserves coproducts.
Precompose both morphisms with every original tensored free
basis inclusion. The actual coproduct comparison identifies
this with the corresponding dual summand. The preceding
lemma and the definition of section evaluation show that both
composites are the same original paired basis morphism.
The unit comparison tensored with the identity equals its right
whiskering by the tensor-identity coherence law; hence the two
original unit comparisons used in the evaluations agree.
These inclusions are jointly epimorphic, proving the full
morphism equality. Tensoring by an invertible line is an
equivalence and preserves epimorphisms; the genuine canonical
pairing is an isomorphism. The stated evaluation is therefore
epimorphic.
\end{proof}

\begin{proposition}[The entire original base ideal is the unit ideal]
\label{prop:generated-line-whole-base-ideal-unit}
\leanok
\lean{EnvelopeProperty.GeneratedLineBaseIdealUnit.baseIdeal_eq_top_of_generatingSections}
For every actually globally generated invertible module $L$
on any scheme $X$, the base ideal generated by ALL original
global sections is $\mathcal O_X$. Empty schemes are included.
\uses{lem:generated-line-dual-evaluation-epi,thm:quasicoherent-image-ideal,prop:actual-invertible-section-family-image}
\end{proposition}
\begin{proof}
The preceding epimorphism has quasicoherent source and target.
On every original affine open its actual section map is
surjective, so its image ideal contains $1$ and is the unit
ideal. This whole image is the sum of the original section
ideals in $G$, each contained in the base ideal generated by
all original global sections. Hence the latter ideal contains
$1$ on every affine open and equals the unit ideal sheaf.
\end{proof}
