\section{The actual projective morphism of an original affine framed family}

\begin{definition}[The original affine projective tuple map]
\label{def:fc-generated-line-projective-affine-map}
\leanok
\lean{EnvelopeProperty.GeneratedLineProjectiveAffineMap.baseCoefficients, EnvelopeProperty.GeneratedLineProjectiveAffineMap.baseCoefficients_naturality, EnvelopeProperty.GeneratedLineProjectiveAffineMap.baseCoefficients_structure, EnvelopeProperty.GeneratedLineProjectiveAffineMap.affineMap}
\uses{lem:fc-original-generated-line-padded-coordinates, thm:fc-original-generator-coefficient-cover}
For an original affine scheme $X$ over $R$, an actual generating
family $G$ of an original line and a genuine whole frame, its actual
finite coefficient tuple, with one zero coordinate, generates the
whole original unit ideal. Use its genuine unit-ideal Rees blowup,
whose actual projection is an isomorphism, and its original generator
projective morphism to obtain an actual map
$X\to\mathbb P_R^{\#G}$.
\end{definition}
\begin{proof}
The preceding actual unit-ideal coefficient producer supplies the
literal spanning condition, without an added premise. The accepted
native affine unit-ideal tuple construction uses the inverse of the
actual unit-ideal blowup projection followed by its actual graded
Rees generator map. Compose with the original whole affine-spectrum
identification of $X$. The actual coefficient homomorphism is that
of the original structure morphism.
\end{proof}

\begin{theorem}[The affine map retains the original structure and is frame independent]
\label{thm:fc-generated-line-projective-affine-frame-independence}
\leanok
\lean{EnvelopeProperty.GeneratedLineProjectiveAffineMap.affineMap_structure, EnvelopeProperty.GeneratedLineProjectiveAffineMap.affineMap_frame_independent}
\uses{def:fc-generated-line-projective-affine-map, thm:fc-original-generated-line-coefficient-naturality}
The actual affine projective tuple map composes with the original
projective structure map to the literal original $X\to\operatorname{Spec}R$.
Changing the genuine original whole line frame leaves the entire
actual scheme morphism unchanged.
\end{theorem}
\begin{proof}
The genuine unit-ideal Rees tuple map preserves its original
coefficient morphism. Native affine-spectrum naturality identifies
that coefficient morphism with the original structure map. For two
actual whole frames, the entire tuple changes by multiplication by
one internally derived whole-ring unit. The actual native Rees
projective generator morphism is invariant under this unit scaling,
so the whole scheme morphisms agree.
\end{proof}

\begin{theorem}[Actual affine pullback naturality]
\label{thm:fc-generated-line-projective-affine-naturality}
\leanok
\lean{EnvelopeProperty.GeneratedLineProjectiveAffineMap.affineMap_naturality}
\uses{thm:fc-generated-line-projective-affine-frame-independence}
Under any actual morphism of affine schemes, pulling back the
original family and actual frame gives exactly the original whole
projective tuple morphism composed with that same literal map.
No projective-map or polarization comparison datum is supplied.
\end{theorem}
\begin{proof}
The actual pulled coefficient family is the original ring map
applied to the entire original coefficient family. This equality
commutes with the actual finite relabeling and the added zero
coordinate. Original affine-spectrum naturality and the genuine
native Rees tuple coefficient-base-change identity therefore give
the entire scheme equality.
\end{proof}
