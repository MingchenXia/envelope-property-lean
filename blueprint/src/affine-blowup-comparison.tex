\section{The affine Rees blowup and the scheme blowup}

\begin{lemma}[Uniqueness of actual blowups]\label{lem:sc0-blowup-universal-iso}
\lean{EnvelopeProperty.Char0.exists_iso_of_isBlowUp}
\leanok
Two morphisms to the same scheme satisfying the blowup universal property for the same ideal
sheaf are isomorphic over that scheme.
\end{lemma}
\begin{proof}
Each morphism is admissible, so the other's universal property gives a map between their
sources over the base. Each composite is an endomorphism over the base, hence equals the
identity by the uniqueness clause of the same universal property. These maps are inverse
isomorphisms, and their construction supplies the compatibility with the base morphisms.
\end{proof}

\begin{lemma}[The affine Rees blowup is the scheme blowup]\label{lem:sc0-affine-global-blowup-iso}
\lean{EnvelopeProperty.Char0.affineBlowUp_isBlowUp,EnvelopeProperty.Char0.exists_affineBlowUp_global_iso,EnvelopeProperty.Char0.idealOfSpecSheaf,EnvelopeProperty.Char0.specIdealSheaf_idealOfSpecSheaf,EnvelopeProperty.Char0.exists_affineBlowUp_center_iso,EnvelopeProperty.Char0.idealOfSpecSheaf_ne_bot_of_blowUp_point}
\leanok
\uses{lem:sc0-blowup-universal-iso,lem:sc0-actual-rees-chart-cover}
For every commutative ring $R$ and ideal $I\subseteq R$, the actual Rees-Proj affine blowup
is isomorphic over $\operatorname{Spec}R$ to the actual scheme blowup of the ideal sheaf
associated to $I$. Every ideal sheaf on $\operatorname{Spec}R$ is of this form, with its ideal
obtained by transporting global sections through the canonical isomorphism
$\Gamma(\operatorname{Spec}R,\mathcal O)\simeq R$.
\end{lemma}
\begin{proof}
The inverse image of $I$ on the Rees-Proj blowup is invertible on its actual affine charts,
where it is generated by the image of the denominator, a nonzerodivisor. The checked affine
universal property gives existence and uniqueness of every admissible lift. The scheme blowup
has the same checked universal property. Apply the preceding uniqueness lemma.
For an arbitrary ideal sheaf on the affine scheme, use the equivalence between quasi-coherent
ideal sheaves and ideals of its ring of global sections, and transport the ideal through the
canonical spectrum global-sections isomorphism. Transporting back gives exactly the original
ideal sheaf, so no new affine-centre hypothesis is needed in applying the comparison.
In particular, if the blowup has an actual point, its normalized ideal cannot be zero:
the comparison would otherwise give a point in the empty affine blowup of the zero ideal.
\end{proof}
