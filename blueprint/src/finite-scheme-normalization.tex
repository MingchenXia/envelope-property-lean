\section{Finite and proper actual normalization of original finite-type models}

\begin{theorem}[Actual original normalization projection is finite]
\label{thm:actual-finite-type-scheme-normalization-finite}
\lean{EnvelopeProperty.IntegralSchemeNormalization.projection_isFinite}
\leanok
\uses{thm:actual-scheme-normalization-finite}
Let an original integral scheme admit an actual locally finite-type
morphism to the spectrum of ANY original field. Its SAME actual
original normalization in its own original function field has a
finite original projection to the original scheme. No perfectness,
Noetherian, quasi-compactness, separatedness or dimension hypothesis
is imposed.
\end{theorem}
\begin{proof}
Directly apply the already proved arbitrary-field theorem for the
SAME original integral scheme, its SAME locally finite-type structure
morphism, and its SAME original generic-field normalization projection.
No affine closure or chart-local finiteness argument is reconstructed.
The resulting finite morphism is the original projection from the
normalization in the original function field.
\end{proof}

\begin{corollary}[Actual normalization projection and proper models]
\label{cor:actual-finite-type-scheme-normalization-proper}
\lean{EnvelopeProperty.IntegralSchemeNormalization.projection_isProper}
\lean{EnvelopeProperty.IntegralSchemeNormalization.projection_comp_isProper}
\leanok
\uses{thm:actual-finite-type-scheme-normalization-finite}
This actual original normalization projection is proper. For any
actual proper model structure on the original integral scheme,
its composite with the actual normalization projection is also
proper over the same original base.
\end{corollary}
\begin{proof}
Every genuine finite original morphism is proper. Proper original
morphisms compose, so compose the resulting actual normalization
projection with the original given proper model structure.
\end{proof}
