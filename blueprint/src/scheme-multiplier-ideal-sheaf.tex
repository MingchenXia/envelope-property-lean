\section{The full multiplier ideal sheaf on the original smooth integral scheme}

\begin{definition}[Actual original global multiplier ideal sheaf]\label{def:sc0-original-global-multiplier-ideal-sheaf}
\lean{EnvelopeProperty.Char0.schemeMultiplierIdealSheaf}
\leanok
\uses{def:sc0-original-scheme-multiplier-affine-ideal,thm:sc0-original-scheme-multiplier-principal}
Let $f:X\to\operatorname{Spec} k$ be the original arbitrary smooth
integral scheme in characteristic zero. For an actual original
ideal sheaf $I$ with nonzero actual generic stalk and rational $q$,
its full multiplier ideal sheaf is the actual scheme ideal sheaf
whose value at every original affine open is the full intrinsic
affine multiplier ideal constructed above.
\end{definition}
\begin{proof}
The preceding complete original basic-open compatibility theorem
is precisely the actual ideal-sheaf restriction axiom. Thus the
whole original affine family is itself actual scheme ideal-sheaf
data. This uses the full computed affine ideals directly; it does
not replace them by a largest compatible subfamily.
\end{proof}

\begin{theorem}[The original affine values of the global construction]\label{thm:sc0-original-global-multiplier-ideal-affine-values}
\lean{EnvelopeProperty.Char0.schemeMultiplierIdealSheaf_ideal}
\leanok
\uses{def:sc0-original-global-multiplier-ideal-sheaf}
At every original affine open $U$, the actual global sheaf has
exactly its full intrinsic affine multiplier ideal, defined by the
original generic germ into the original function field and every
actual divisorial DVR contribution.
\end{theorem}
\begin{proof}
This is the actual affine component of the preceding constructor.
\end{proof}

\begin{theorem}[Actual finite presentation of the whole original multiplier sheaf]\label{thm:sc0-original-global-multiplier-module-finite-presentation}
\lean{EnvelopeProperty.Char0.schemeMultiplierIdealSheaf_isFinitePresentation}
\leanok
\uses{def:sc0-original-global-multiplier-ideal-sheaf}
The actual module sheaf of this whole original global multiplier
ideal sheaf is of finite presentation on $X$.
\end{theorem}
\begin{proof}
The original smooth structure morphism is locally of finite type
over the actual Noetherian base field. Hence the whole original
scheme is locally Noetherian. The already proved actual
ideal-sheaf finite-presentation theorem applies to this genuine
ideal sheaf: its original affine ideal modules are finite over
Noetherian section rings, hence finitely presented, and their
actual restrictions are the corresponding localized modules.
Their affine-local finite presentations give finite presentation
of the whole actual module sheaf.
\end{proof}
