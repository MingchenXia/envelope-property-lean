\section{Actual original normal proper simultaneous PL models}

\begin{definition}[The true original normalized simultaneous model]
\label{def:actual-original-normalized-simultaneous-model}
\lean{EnvelopeProperty.ProjSubvariety.normalizedSimultaneousIdealBlowup}
\lean{EnvelopeProperty.ProjSubvariety.normalizedSimultaneousIdealBlowupProjection}
\lean{EnvelopeProperty.ProjSubvariety.normalizedSimultaneousIdealBlowupStructure}
\leanok
\uses{thm:actual-original-pl-simultaneous-blowup-integral}
For two original ideals whose model logarithms are finite at every original
Gauss point, take the actual generic-field normalization of their same actual
simultaneous blowup, its actual original normalization projection, and its
same original polynomial-base structure morphism.
\end{definition}
\begin{proof}
At the original trivial analytic point, finite ideal logarithms construct an
actual Gauss lift on the original simultaneous blowup, hence prove its
integrality by the preceding theorem. Use this derived integral structure
in the actual native generic-field normalization. Compose its actual
projection with the same original blowup structure morphism.
\end{proof}

\begin{theorem}[Actual integrality and normal original local rings]
\label{thm:actual-original-normalized-simultaneous-model-normal}
\lean{EnvelopeProperty.ProjSubvariety.normalizedSimultaneousIdealBlowup_stalk_isIntegrallyClosed}
\lean{EnvelopeProperty.ProjSubvariety.normalizedSimultaneousIdealBlowup_isIntegral}
\leanok
\uses{def:actual-original-normalized-simultaneous-model,thm:actual-integral-scheme-normalization-normal-stalks}
This same actual normalized model is integral, and every one of its actual
original local rings is integrally closed.
\end{theorem}
\begin{proof}
The native relative normalization of an integral source is integral; here
that source is the original generic-field spectrum. The proved original
normalization-stalk theorem, via actual integral-closure charts and actual
prime localizations, makes every original local ring integrally closed.
No normal-model or chart-normality certificate is supplied.
\end{proof}

\begin{theorem}[Actual finite normalization and proper original model]
\label{thm:actual-original-normalized-simultaneous-model-proper}
\lean{EnvelopeProperty.ProjSubvariety.normalizedSimultaneousIdealBlowupProjection_isFinite}
\lean{EnvelopeProperty.ProjSubvariety.normalizedSimultaneousIdealBlowupStructure_isProper}
\leanok
\uses{def:actual-original-normalized-simultaneous-model,def:actual-original-pl-product-field-structure,cor:actual-finite-type-scheme-normalization-proper}
Over any original coefficient field, the actual normalization
projection is finite and the same normalized model is proper over the
original polynomial base.
\end{theorem}
\begin{proof}
The actual original simultaneous blowup's structure over its original
coefficient field is locally of finite type by the preceding genuine field
structure theorem. Finite normalization over the original field therefore proves
finiteness of its actual original normalization projection. A finite
morphism is proper; compose it with the actual proper original simultaneous
blowup structure over the same original polynomial base. Thus all geometric
model properties asserted here are derived from the original data.
\end{proof}
