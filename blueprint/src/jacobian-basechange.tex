\section{Base change and spreading of the actual Jacobian image ideal}

\begin{lemma}[The actual Jacobian image ideal commutes with differential base change]\label{lem:sc0-jacobian-image-ideal-basechange}
\lean{EnvelopeProperty.Char0.map_affineRelativeJacobianIdeal}
\leanok
\uses{lem:sc0-relative-differential-det-basechange,lem:sc0-affine-jacobian-principal}
Given actual source \'etale coordinates on $R$, an actual differential basis on
$A_0$, and a formally \'etale map $A_0\to A$, the actual Jacobian image ideal in
the transported differential basis on $A$ is the extension of the original image
ideal from $A_0$.
\end{lemma}
\begin{proof}
Both actual image ideals are principal with the actual relative determinant as
generator. The determinant base-change identity identifies their generators
under $A_0\to A$, and extension of a principal ideal is generated by the image
of its generator.
\end{proof}

\begin{lemma}[The actual monomial Jacobian ideal spreads from a localized stalk]\label{lem:sc0-jacobian-monomial-spreading}
\lean{EnvelopeProperty.Char0.exists_away_affineRelativeJacobianIdeal_eq_monomial}
\leanok
\uses{lem:sc0-jacobian-image-ideal-basechange}
Suppose $A_0$ is Noetherian, $A$ is its localization at a multiplicative set $M$,
and in the transported differential basis its actual Jacobian image ideal equals
the ideal generated by the image of a monomial $\prod_i p_i^{a_i}$ from $A_0$.
There exists $s\in M$ such that on $A_0[1/s]$ the actual original Jacobian image
ideal is generated by the same image monomial.
\end{lemma}
\begin{proof}
The base-change lemma gives equality after localization of the original Jacobian
ideal and the principal monomial ideal. Both are finitely generated by
Noetherianity. Clear the finitely many denominators in these two inclusions;
their product $s\in M$ makes the two extended ideals equal after inverting $s$.
The image of the monomial ideal is generated by the image of its monomial.
\end{proof}
