\section{Original-source localization and actual multiplier contributions}

\begin{theorem}[Source localization preserves the intrinsic DVR canonical ideal]\label{thm:sc0-intrinsic-canonical-source-localization}
\lean{EnvelopeProperty.Char0.intrinsicDVRSourceJacobianIdeal_source_localization}
\leanok
For actual $k$-subalgebras $C\subseteq E\subseteq V\subset K$, where
$E$ is an actual localization of $C$ and $V$ is an actual essentially
finite-type divisorial DVR, the intrinsic original-source DVR Jacobian
ideals of $C$ and $E$ are identical ideals of $V$.
\end{theorem}
\begin{proof}
Produce actual finite DVR coordinates internally. Independence from
every finite presentation identifies both intrinsic ideals with the
corresponding actual derivative determinant ideals in these same
coordinates. The actual source-localization spanning theorem says
that original-source differentials span the entire localized
differential module. By alternating multilinearity, the derivative
ideals generated by every original source tuple and by every localized
tuple are therefore equal. Their intrinsic coordinate-independent
ideals agree as asserted.
\end{proof}

\begin{corollary}[Exact pullback of each actual localized contribution]\label{cor:sc0-intrinsic-multiplier-source-localization}
\lean{EnvelopeProperty.Char0.intrinsicDVRMultiplierContribution_source_localization}
\leanok
\uses{thm:sc0-intrinsic-canonical-source-localization}
For an original nonzero ideal $I\subset C$ and rational exponent $q$,
its actual $C$-contribution at $V$ is the comap of the actual
$E$-contribution for $IE$ along the actual map $C\to E$.
\end{corollary}
\begin{proof}
The two canonical Jacobian ideals coincide by the preceding theorem.
Extension of the original ideal through $C\to E\to V$ equals its
direct actual extension $C\to V$, so the exact cutoff exponents
coincide. The defining comaps compose by the same actual inclusion
maps, giving equality of the original-source contribution ideals.
\end{proof}

\begin{theorem}[Localization maps the full ideal into the localized full ideal]\label{thm:sc0-multiplier-localization-forward}
\lean{EnvelopeProperty.Char0.map_valuativeMultiplierIdeal_le_source_localization}
\leanok
\uses{cor:sc0-intrinsic-multiplier-source-localization}
When $K$ is the original fraction field of $C$, actual extension to
$E$ maps $\mathcal J_C(I^q)$ into $\mathcal J_E((IE)^q)$.
\end{theorem}
\begin{proof}
Every actual divisorial DVR containing $E$ contains $C$, so full
original-source membership gives its contribution at every such DVR.
The exact contribution comparison converts this into the actual
localized-source contribution of the image of the original element.
Thus the image of every original multiplier element belongs to the
full localized-source ideal. Ideal extension is generated by those
images, so the entire extended ideal is contained in it.
\end{proof}
