\section{Original-source determinant ideals under actual localization}

\begin{theorem}[Localization preserves all source-function derivative determinants]\label{thm:sc0-derivation-source-localization}
\lean{EnvelopeProperty.Char0.derivationSourceJacobianIdeal_source_localization}
\leanok
\uses{lem:sc0-multilinear-submodule-from-generators}
Let $S=M^{-1}R$ be an actual localization of $k$-algebras, let $A$ be a
compatible $S$-algebra, and let $D_1,\ldots,D_n:S\to A$ be actual
$k$-derivations. The ideal of $A$ generated by determinants
$\det(D_j(r_i))$ for original tuples $r_i\in R$ is equal to the ideal
generated by determinants for all tuples $s_i\in S$.
\end{theorem}
\begin{proof}
Every original tuple is also a localized tuple, giving one inclusion.
For the reverse inclusion, the original differentials
$d(r)$ span $\Omega_{S/k}$ over $S$, because $S$ is the actual
localization of $R$. The determinant of the linear forms induced by
the $D_j$ is an $S$-multilinear map on $n$ copies of
$\Omega_{S/k}$. Compose it with the quotient by the ideal generated
by original tuples. The resulting multilinear map vanishes on all
tuples from the spanning family, hence vanishes identically by
multilinearity. Evaluating it at $d(s_i)$ shows that every localized
tuple determinant belongs to the original determinant ideal.
\end{proof}
