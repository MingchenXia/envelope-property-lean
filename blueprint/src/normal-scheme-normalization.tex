\section{Normality of the actual original integral-scheme normalization}

\begin{theorem}[Every original normalization stalk is integrally closed]
\label{thm:actual-integral-scheme-normalization-normal-stalks}
\lean{EnvelopeProperty.IntegralSchemeNormalization.stalk_isIntegrallyClosed}
\leanok
\uses{scheme:actual-integral-normalization-affine-charts}
For every original integral scheme, every actual local stalk of
its genuine original normalization in its original function field
is integrally closed. No finite-type, perfect-field, finite
normalization, or Noetherian assumption is required.
\end{theorem}
\begin{proof}
For an actual point of the original normalization, choose an
actual nonempty affine neighborhood of its original projection.
The true original normalization chart has as its entire original
image the inverse image of that neighborhood, so the point is
actually the image of a point on the literal original affine
normalization chart. Its coordinate ring is the actual integral
closure of the original affine domain in its own original fraction
field, hence is an integrally closed domain. The actual localization
at the chart point's original prime ideal remains integrally
closed. The native original spectrum stalk isomorphism identifies
that localization with the actual spectrum stalk. Finally the
actual chart open immersion induces a genuine original stalk
isomorphism to the original normalization stalk. Transport integral
closedness through these two actual original ring isomorphisms.
\end{proof}

\begin{corollary}[Standard geometric unibranchness of the original normalization]
\label{cor:actual-integral-scheme-normalization-unibranch}
\lean{EnvelopeProperty.IntegralSchemeNormalization.normalization_isGeometricallyUnibranch}
\leanok
\uses{thm:actual-integral-scheme-normalization-normal-stalks}
The actual original normalization has the standard structural
geometric unibranch property at every actual original local stalk.
\end{corollary}
\begin{proof}
The genuine original normalization is integral, so all actual
original stalks are domains. Apply the preceding integral
closedness theorem and the proved original normal-local-ring
geometric-unibranch theorem at each actual original stalk.
\end{proof}
