\label{sec:witt-two-affine-residue-geometry}

\begin{theorem}[Actual affine special-fibre isomorphism]
\label{thm:witt-two-actual-affine-special-fibre-iso}
\lean{EnvelopeProperty.DeligneIllusie.WittTwoAffineResidueGeometry.fibre,EnvelopeProperty.DeligneIllusie.WittTwoAffineResidueGeometry.residueAffineIso}
\leanok
The actual affine special fibre is canonically isomorphic to
$\operatorname{Spec}(k\otimes_{W_2(k)}\Gamma(\widetilde X,U))$
with its SAME original structural coefficient actions.
\uses{def:witt-two-scheme-affine-section-restriction}
\end{theorem}
\begin{proof}
The actual affine-open spectrum isomorphism identifies $U$ with
the spectrum of its original section ring. Its composite with
the spectrum of the ORIGINAL Witt structural section map is
exactly the given map from $U$ to the Witt base: this follows
from the actual affine spectrum section map of the original
structure morphism and the spectrum global-sections equivalence.
Thus the actual affine-open isomorphism induces an isomorphism
on the two actual categorical pullbacks. The genuine affine
spectrum tensor-product pullback isomorphism then gives exactly
the asserted original tensor ring. All comparison maps are
constructed from the original arrows by the pullback universal
property.
\end{proof}

\begin{theorem}[Both original geometric projections are retained]
\label{thm:witt-two-actual-affine-special-fibre-projections}
\lean{EnvelopeProperty.DeligneIllusie.WittTwoAffineResidueGeometry.residueAffineIso_fst,EnvelopeProperty.DeligneIllusie.WittTwoAffineResidueGeometry.residueAffineIso_snd}

projection to $\operatorname{Spec}k$ is the spectrum of the
original left coefficient map. The actual projection to $U$,
followed by its original affine spectrum isomorphism, is the
spectrum of the original right tensor inclusion.
\uses{thm:witt-two-actual-affine-special-fibre-iso}
\end{theorem}
\begin{proof}
The actual pullback map defining the first comparison preserves

The spectrum tensor-product pullback isomorphism also preserves
the original left and right inclusions. Compose these equations
and cancel the native isomorphism inverses to obtain the two
asserted entire scheme-morphism equalities.
\end{proof}

\label{thm:witt-two-actual-affine-special-fibre-section-ring}

\uses{thm:witt-two-actual-affine-special-fibre-iso,thm:witt-two-actual-affine-special-fibre-projections}

\begin{proof}
