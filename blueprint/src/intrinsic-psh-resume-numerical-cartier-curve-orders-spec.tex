\section{Native spectrum coefficients and principal Cartier weights}

\begin{definition}[The native spectrum residue isomorphism preserves original coefficients]
\label{def:intrinsic-numerical-spec-native-residue-algebra}
\uses{def:fc-affine-scheme-local-coefficients}
\lean{EnvelopeProperty.NumericalCartierCurveOrders.specResidueAlgEquiv}
\leanok
Let $A$ be an original commutative $k$-algebra and $x\in\operatorname{Spec}A$.
The native spectrum residue isomorphism
$\kappa(x)\simeq\kappa(A_x)$ is an actual $k$-algebra isomorphism for
the original structural map induced by $k\to A$.
\end{definition}
\begin{proof}
The original scalar $c\in k$ becomes the actual global section given
by $c\in A$. Naturality of the canonical spectrum global-section
isomorphism identifies this section with the actual pullback of the
original scalar. The native spectrum residue-isomorphism triangle
sends its actual germ and residue to the image of that same $c$ in
$\kappa(A_x)$. Thus the existing whole native residue isomorphism
commutes with every original $k$-coefficient.
\end{proof}

\begin{theorem}[Native principal Cartier weight on the original spectrum]
\label{thm:intrinsic-numerical-spec-principal-cartier-weight}
\uses{def:fc-curve-cartier-original-equations,def:intrinsic-numerical-spec-native-residue-algebra,thm:fc-curve-cartier-native-local-orders,thm:fc-scheme-order-local-length,thm:fc-original-principal-ring-iso-order,thm:fc-curve-proper-support-points}
\lean{EnvelopeProperty.NumericalCartierCurveOrders.weightedLocalOrder_principal_spec}
\leanok
Let $A$ be an original Noetherian domain of dimension at most one,
with its original $k$-algebra structure. Let $I$ be an actual Cartier
ideal on $\operatorname{Spec}A$ generated by the actual global section
corresponding to a nonzero $a\in A$. For every original nonzero prime
$x$, its native weighted Cartier order is
\[
 \operatorname{ord}_x(I)\operatorname{rk}_k\kappa(x)
 =\operatorname{ord}_{A_x}(a)\operatorname{rk}_k\kappa(A_x).
\]
Both residue weights are the original natural ranks used in the
weighted-order definition; this local identity does not assert residue
finiteness.
The equation is original geometric input; no order or residue comparison
is assumed.
\end{theorem}
\begin{proof}
The nonzero original prime has coheight one. The actual global equation
therefore computes the native Cartier order as the length of the same
principal stalk quotient. The native spectrum stalk isomorphism takes
its germ to the actual image of $a$ in $A_x$, by the original
global-section--germ triangle. The genuine ring-isomorphism order
theorem identifies the principal orders. The preceding actual residue
algebra isomorphism identifies their original $k$-degrees. Multiplying
these same-point identities proves the formula.
\end{proof}
