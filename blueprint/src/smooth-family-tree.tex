\section{Simultaneous family trees over an arbitrary smooth source}

\begin{lemma}[Original source data through a localization]\label{lem:sc0-family-leaf-source-localization}
\lean{EnvelopeProperty.Char0.EtaleFamilyMonomialLeaf.of_source_localization}
\leanok
\uses{def:sc0-etale-family-monomial-leaf,lem:sc0-jacobian-source-localization}
Let $C\subseteq D\subseteq E$ be literal subalgebras of one field, with $D$
an actual localization of $C$. If the extensions to $E$ of the ideals
$\mathfrak a_jD$ and the all-$D$-functions Jacobian ideal are monomial in one
family of actual \'etale coordinates on $E$, then the original ideals
$\mathfrak a_j$ and the all-$C$-functions Jacobian ideal have exactly those
same monomial expressions in those same coordinates.
\end{lemma}
\begin{proof}
Extension of ideals is functorial, and the two literal inclusions compose to
the original inclusion $C\hookrightarrow E$. This identifies every factor
ideal. The genuine Jacobian ideal is unchanged when the source is localized:
original source differentials span the module of differentials of the
localized source, and the determinant is multilinear, so its image ideal is
already generated by the original source differentials. Apply that equality
in the actual target coordinate basis and retain the supplied monomial formula.
\end{proof}

\begin{theorem}[A smooth source produces a simultaneous family Rees tree]\label{thm:sc0-smooth-family-rees-tree}
\lean{EnvelopeProperty.Char0.exists_smooth_family_rees_tree}
\leanok
\uses{lem:sc0-family-leaf-source-localization,thm:sc0-bp-family-monomial-tree,cor:sc0-smooth-source-unit-rees-etale-cover}
For every smooth literal affine domain $C$ over a characteristic-zero field
and every finite family of nonzero ideals $\mathfrak a_j\subset C$, there
exist an integer $n$ and a literal finite Rees chart tree rooted at $C$ such
that every leaf has actual \'etale coordinates of dimension $n$ in which all
original factor ideals and the genuine all-original-$C$-functions Jacobian
ideal are monomial in one finite injective selection of boundary coordinates.
\end{theorem}
\begin{proof}
The irreducible smooth source has one pure relative dimension $n$. Its actual
adapted coordinate neighbourhoods admit a finite principal refinement. Choose
the finite generators $g_i$ of the unit ideal defining that refinement. The
literal Rees algebra chart for every nonzero $g_i$ is the actual localization
$C[g_i^{-1}]$ under its original inclusion, and has actual \'etale source
coordinates. Each extended factor remains nonzero because the inclusion into
the original field is injective. Apply the actual BP family-product tree
producer separately on each of these literal source charts. Attach the
resulting actual child trees to the unit-generating Rees node rooted at $C$.
Every leaf is a leaf of one actual child tree. Apply the preceding localization
lemma to its simultaneous coordinates: all ideal extensions compose under
the literal inclusions, and its Jacobian ideal is the ideal generated by all
original $C$-functions. These identities give exactly the asserted leaf data.
The tree is finite because its root has finitely many nonzero-generator
branches and each child tree is finite.
\end{proof}
