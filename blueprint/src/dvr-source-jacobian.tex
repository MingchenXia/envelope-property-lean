\section{Finite positive actual original-source discrepancy}

\begin{lemma}[The genuine field Jacobian ideal of a localized source is the unit ideal]\label{lem:sc0-localized-source-field-jacobian-unit}
\lean{EnvelopeProperty.Char0.affineRelativeJacobianIdeal_self_eq_top_of_coordinate_basis, EnvelopeProperty.Char0.affineRelativeJacobianIdeal_localization_eq_top_of_coordinate_basis}
\leanok
\uses{lem:sc0-jacobian-source-localization}
If a field $K$ is the actual fraction field of $C$, the genuine ideal
of all original $C$-function Jacobians in any actual differential coordinate
basis of $K$ is the unit ideal.
\end{lemma}
\begin{proof}
Actual source localization preserves the genuine all-functions Jacobian
ideal. After replacing the source by $K$, the tuple consisting of the
actual basis coordinate functions has the identity differential matrix.
Its determinant is one and is an actual generator of the ideal. Therefore
the ideal is the unit ideal, retaining all original $C$-functions by the
proved source-localization equality.
\end{proof}

\begin{theorem}[The actual original-source DVR Jacobian ideal is nonzero]\label{thm:sc0-original-source-dvr-jacobian-nonzero}
\lean{EnvelopeProperty.Char0.dvrFieldDifferentialBasis, EnvelopeProperty.Char0.map_dvrOriginalSourceJacobianIdeal, EnvelopeProperty.Char0.dvrOriginalSourceJacobianIdeal_ne_bot}
\leanok
\uses{lem:sc0-localized-source-field-jacobian-unit,def:sc0-original-source-dvr-jacobian-ideal,lem:c0-dvr-jacobian}
For an actual divisorial DVR containing an actual domain $C$ in its actual
fraction field $K$, the genuine all-original-$C$-functions DVR Jacobian
ideal is nonzero. No global source coordinate family is assumed.
\end{theorem}
\begin{proof}
The actual DVR field coordinates give an actual differential basis of
$\Omega_{K/k}$ with the corresponding finite coordinate index. Extending
the actual DVR original-source Jacobian ideal to $K$ maps every actual
derivative determinant to its determinant in that actual field basis.
The source-function matrix in the field basis is the transpose of the
mapped actual DVR derivative matrix, so their determinants agree exactly.
Thus the extended ideal is exactly the original-source field Jacobian
ideal, which is the unit ideal by the preceding lemma. The zero ideal
extends to zero; since the actual field is nontrivial, the original ideal
cannot be zero.
\end{proof}

\begin{corollary}[The actual original-source discrepancy is finite]\label{cor:sc0-original-source-dvr-jacobian-finite-order}
\lean{EnvelopeProperty.Char0.dvrIdealOrder_ne_top_of_ne_bot, EnvelopeProperty.Char0.dvrOriginalSourceJacobianIdeal_order_ne_top}
\leanok
\uses{thm:sc0-original-source-dvr-jacobian-nonzero,lem:sc0-dvr-ideal-order-principal-product}
The actual original-source DVR Jacobian ideal has finite normalized order.
Consequently one plus its normalized order is a finite positive integer.
\end{corollary}
\begin{proof}
An actual DVR ideal is generated by its actual principal generator. Its
order is the normalized valuation of that generator. Nonzero ideal means
nonzero generator, whose actual normalized valuation is finite. Adding one
preserves finiteness and gives strict positivity.
\end{proof}
